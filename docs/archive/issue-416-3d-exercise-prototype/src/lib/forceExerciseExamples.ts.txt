import type { ForceMeasurementMode } from "./forceSetup";

export type Point3 = readonly [x: number, y: number, z: number];

export type BodyJoint =
  | "neck"
  | "leftShoulder"
  | "leftElbow"
  | "leftWrist"
  | "rightShoulder"
  | "rightElbow"
  | "rightWrist"
  | "leftHip"
  | "leftKnee"
  | "leftAnkle"
  | "rightHip"
  | "rightKnee"
  | "rightAnkle";

export type BodyPose = Record<BodyJoint, Point3>;

export type ForceExerciseFamily = "block_pull" | "rotator_cuff" | "hip_rotator";

export interface ForceExerciseModePresentation {
  description: string;
  actionCue: string;
  startPose: BodyPose;
  endPose: BodyPose;
}

  setupCue: string;

  attachmentJoint: BodyJoint;
  originLabel: string;
  originJoint: BodyJoint;
  anchor: Point3;
  modes: Record<ForceMeasurementMode, ForceExerciseModePresentation>;
  camera: Point3;
  cameraTarget: Point3;
}

function staticPresentation(
  description: string,
  actionCue: string,
  staticPose: BodyPose,
): ForceExerciseModePresentation {
  return { description, actionCue, startPose: staticPose, endPose: staticPose };
}

function movementPresentation(
  description: string,
  actionCue: string,
  startPose: BodyPose,
  endPose: BodyPose,
): ForceExerciseModePresentation {
  return { description, actionCue, startPose, endPose };
}

function normalized(value: string): string {
  return value.trim().toLocaleLowerCase().replace(/[^a-z0-9]+/g, " ");
}

export function suggestedExerciseFamily(exercise: string): ForceExerciseFamily {
  const value = normalized(exercise);
  if (/\b(hip|glute|piriformis)\b/.test(value)) return "hip_rotator";
  if (/\b(rotator|cuff|shoulder|external rotation|internal rotation)\b/.test(value)) {
    return "rotator_cuff";
  }
  return "block_pull";
}

export function examplesForFamily(family: ForceExerciseFamily): ForceExerciseExample[] {
  return FORCE_EXERCISE_EXAMPLES.filter((example) => example.family === family);
}

export function presentationForMode(
  example: ForceExerciseExample,
  mode: ForceMeasurementMode,
): ForceExerciseModePresentation {
  return example.modes[mode];
}

export function exampleModeCopy(mode: ForceMeasurementMode): {
  label: string;
  instruction: string;
} {
  return mode === "movement"
    ? {
        label: "Movement demonstration",
        instruction: "The resisted distal attachment stays fixed while the highlighted origin-side segment moves. The loop is illustrative, not protocol timing or body tracking.",
      }
    : {
        label: "Isometric setup",
        instruction: "The model holds the working position while the force path remains fixed.",
      };
}
